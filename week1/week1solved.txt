# ----------------------------------------
# CHALLENGE 2: Función que retorna valor
# ----------------------------------------

def sumar(a, b):
    return a + b

resultado = sumar(5, 3)
print("La suma es:", resultado)

# ----------------------------------------
# CHALLENGE 4: Función con validación
# ----------------------------------------

def par_o_impar(numero):
    if numero % 2 == 0:
        print(numero, "es par")
    else:
        print(numero, "es impar")

par_o_impar(8)
par_o_impar(7)


# ----------------------------------------
# CHALLENGE 5: Reutilizar función varias veces

# ----------------------------------------
# CHALLENGE 6: Función con lógica
# ----------------------------------------

def calcular_bono(sueldo):
    if sueldo >= 20000:
        bono = sueldo * 0.10
    else:
        bono = sueldo * 0.05
    return bono

print("Bono:", calcular_bono(25000))
print("Bono:", calcular_bono(15000))


# ----------------------------------------
# CHALLENGE 7: Combinar funciones
# ----------------------------------------

def nombre_completo(nombre, apellido):
    return nombre + " " + apellido

def despedida(nombre):
    print("Hasta luego", nombre)

persona = nombre_completo("Marcos", "Lopez")
saludar(persona)
despedida(persona)
